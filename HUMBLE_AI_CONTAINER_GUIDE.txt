🚀 快速开始（15分钟）| Quick Start
═══════════════════════════════════════════════════════════════════════════

Step 1: 创建项目目录
步骤1：Create Project Directory

mkdir -p ~/robotics_ai_project
cd ~/robotics_ai_project
mkdir -p my_code ros2_ws/src downloads


Step 2: 创建docker-compose.yml
步骤2：Create docker-compose.yml

cat > docker-compose.yml << 'EOF'
version: '3.8'

services:
  ros2_ai_workstation:
    build: 
      context: ./
      dockerfile: Dockerfile
    image: ros2_humble_ai_gesture:v1.0
    container_name: ros2_ai_workstation
    
    # 网络配置（必需）
    network_mode: host
    privileged: true
    ipc: host
    
    # GPU配置（如果有NVIDIA GPU）
    deploy:
      resources:
        reservations:
          devices:
            - driver: nvidia
              count: 1
              capabilities: [gpu]
    
    # 环境变量
    environment:
      - DISPLAY=${DISPLAY}
      - QT_X11_NO_MITSHM=1
      - NVIDIA_VISIBLE_DEVICES=all
      - NVIDIA_DRIVER_CAPABILITIES=all
      - ROS_DOMAIN_ID=30
      - TURTLEBOT3_MODEL=waffle_pi
    
    # 设备映射（摄像头）
    devices:
      - /dev/video0:/dev/video0
    
    # 卷映射
    volumes:
      - /tmp/.X11-unix:/tmp/.X11-unix:rw
      - ./my_code:/root/my_code
      - ./ros2_ws:/root/ros2_ws
      - ./downloads:/root/downloads
    
    stdin_open: true
    tty: true
EOF


Step 3: 创建Dockerfile
步骤3：Create Dockerfile

cat > Dockerfile << 'EOF'
FROM osrf/ros:humble-desktop-full

ENV DEBIAN_FRONTEND=noninteractive
ENV ROS_DISTRO=humble

RUN apt-get update && apt-get install -y \
    python3-pip \
    vim \
    git \
    wget \
    curl \
    tree \
    && rm -rf /var/lib/apt/lists/*

RUN mkdir -p /root/ros2_ws/src

WORKDIR /root

RUN echo "source /opt/ros/humble/setup.bash" >> /root/.bashrc

CMD ["bash"]
EOF


Step 4: 启动容器
步骤4：Start Container

# 允许X11访问
xhost +local:docker

# 构建并启动
docker-compose up -d --build

# 进入容器
docker-compose exec ros2_ai_workstation bash
